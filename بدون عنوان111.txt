// 🧠 HURST EXPONENT CALCULATION - Fractal Dimension Detection
// ═══════════════════════════════════════════════════════════════════════════════════════════
f_hurstExponent(src, length) =>
    mean = ta.sma(src, length)
    var float sumDev = 0.0
    var float maxDev = 0.0
    var float minDev = 0.0
    sumDev := 0.0
    maxDev := 0.0
    minDev := 0.0    
    for i = 0 to length - 1
        deviation = src[i] - mean
        sumDev := sumDev + deviation
        maxDev := math.max(maxDev, sumDev)
        minDev := math.min(minDev, sumDev)    
    range_rs = maxDev - minDev
    stdDev = ta.stdev(src, length)
    rs_ratio = stdDev > 0 ? range_rs / stdDev : 1.0
    hurst = rs_ratio > 0 ? math.log(rs_ratio) / math.log(length) : 0.5
    math.max(0.2, math.min(0.8, hurst))
